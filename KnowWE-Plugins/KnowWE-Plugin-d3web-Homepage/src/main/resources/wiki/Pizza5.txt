!!! General Domain Model

! Classes and properties without individuals 

%%turtle
pizza5:Pizza a rdfs:Class.
pizza5:PizzaBase a rdfs:Class.
pizza5:PizzaTopping a rdfs:Class.

pizza5:hasBase a rdf:Property .
pizza5:hasTopping a rdf:Property .

pizza5:Pizza pizza5:hasBase pizza5:PizzaBase .
pizza5:Pizza pizza5:hasTopping pizza5:PizzaTopping .
%
